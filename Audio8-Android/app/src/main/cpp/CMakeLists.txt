cmake_minimum_required(VERSION 3.22)
project(Audio8Android LANGUAGES C CXX)
include(CheckIPOSupported)
check_ipo_supported(RESULT AUDIO8_IPO)
if(AUDIO8_IPO)
  set(CMAKE_INTERPROCEDURAL_OPTIMIZATION_RELEASE ON)
  set(CMAKE_INTERPROCEDURAL_OPTIMIZATION_RELWITHDEBINFO ON)
endif()
set(CMAKE_C_FLAGS_RELWITHDEBINFO "${CMAKE_C_FLAGS_RELWITHDEBINFO} -O3")
set(CMAKE_CXX_FLAGS_RELWITHDEBINFO "${CMAKE_CXX_FLAGS_RELWITHDEBINFO} -O3")
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(AUDIOCPP_MODEL_SET custom CACHE STRING "" FORCE)
set(AUDIOCPP_MODELS audio8_tts CACHE STRING "" FORCE)
set(AUDIOCPP_DEPLOYMENT_BUILD ON CACHE BOOL "" FORCE)
set(AUDIOCPP_BUILD_C_API ON CACHE BOOL "" FORCE)
set(ENGINE_ENABLE_NATIVE_CPU OFF CACHE BOOL "" FORCE)
set(ENGINE_ENABLE_OPENMP OFF CACHE BOOL "" FORCE)
set(ENGINE_ENABLE_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(ENGINE_ENABLE_CUDA OFF CACHE BOOL "" FORCE)
set(ENGINE_ENABLE_METAL OFF CACHE BOOL "" FORCE)
set(ENGINE_ENABLE_VULKAN OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ARM_ARCH "armv8.6-a+dotprod+fp16+i8mm" CACHE STRING "" FORCE)
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/audio8-optimization.txt"
  CONTENT "build_type=${CMAKE_BUILD_TYPE}\narm_arch=${GGML_CPU_ARM_ARCH}\nipo_supported=${AUDIO8_IPO}\nrelwithdebinfo_lto=${CMAKE_INTERPROCEDURAL_OPTIMIZATION_RELWITHDEBINFO}\nrelwithdebinfo_cxx_flags=${CMAKE_CXX_FLAGS_RELWITHDEBINFO}\n")
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ALL_VARIANTS OFF CACHE BOOL "" FORCE)
set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} -march=armv8.6-a+dotprod+fp16+i8mm")
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -march=armv8.6-a+dotprod+fp16+i8mm")
set(AUDIO8_RUNTIME "${CMAKE_CURRENT_SOURCE_DIR}/../../../../runtime")
if(NOT EXISTS "${AUDIO8_RUNTIME}/include/audiocpp.h")
  message(FATAL_ERROR "Pinned audio.cpp runtime missing; run scripts/prepare-runtime.sh")
endif()
add_subdirectory("${AUDIO8_RUNTIME}" "${CMAKE_CURRENT_BINARY_DIR}/runtime" EXCLUDE_FROM_ALL)
target_link_libraries(audiocpp PRIVATE log)
target_link_options(audiocpp PRIVATE "-Wl,-z,max-page-size=16384")
add_library(audio8_jni SHARED audio8_jni.cpp)
target_include_directories(audio8_jni PRIVATE "${AUDIO8_RUNTIME}/include")
target_link_libraries(audio8_jni PRIVATE audiocpp log)
target_link_options(audio8_jni PRIVATE "-Wl,-z,max-page-size=16384")
